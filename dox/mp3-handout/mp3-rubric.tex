\section*{Checkpoint 1 (0-30pts)}

\begin{enumerate}
    \item Cache (6pts)
    \begin{itemize}
        \item Cache is empty on OS startup
        \item Starting with an empty cache, can induce 64 misses on different blocks and then 64 read hits on those same blocks with a total of 64 requests made to the backing device
        \item Must not generate a request to the backing device on a read hit
        \item No evictions unless \texttt{cache\_get\_block} is called again
    \end{itemize}
    \item KTFS (10pts)
    \begin{itemize}
        \item Open/Close
        \item Readat
        \item Ioctl (GETEND)
    \end{itemize}
    \item Vioblk (7pts)
    \begin{itemize}
        \item Open/Close
        \item Readat
        \item Writeat
        \item Ioctls (GETEND, GETBLKSZ)
    \end{itemize}
    \item ELF Loader (4pts)
    \begin{itemize}
        \item Load a valid ELF file into appropriate memory and return the entry point
        \item Reject invalid ELFs with an error code (magic bytes, class, endianness, version, type, architecture, etc.)
    \end{itemize}
    \item Memio (2pt)
    \begin{itemize}
        \item Readat
        \item Writeat
        \item Ioctls (GETEND, GETBLKSZ, SETEND)
    \end{itemize}
    \item Locks (1pt)
\end{enumerate}

\subsection*{Penalties}

\begin{enumerate}
    \item Unable to check out \texttt{main} branch (-30pts)
    \item Unable to compile \texttt{main} branch (-30pts)
    \item Compiler warnings when compiling \texttt{kernel.elf} (-5pts)
\end{enumerate}
